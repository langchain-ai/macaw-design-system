export const matchesAxisValues = (axes, values) =>
  Object.entries(axes).every(([key, axis]) =>
    axis.options.some((option) =>
      typeof option === 'string'
        ? option === values[key]
        : option.value === values[key]
    )
  );

export const defineEntry = (entry) => {
const baseEntry = {
name: entry.name,
tag: entry.tag ?? toTag(entry.name),
category: entry.category,
storybookTitle:
entry.storybookTitle ?? toStorybookTitle(entry.name, entry.category),
bodyClassName: entry.bodyClassName,
};
if (!('axes' in entry)) {
return { ...baseEntry, axes: {}, render: entry.render };
}
return {
...baseEntry,
axes: entry.axes,
render: (values) =>
matchesAxisValues(entry.axes, values) ? entry.render(values) : null,
};
};

export const normalizeOptions = (axis) =>
  axis.options.map((option) =>
    typeof option === 'string' ? { value: option, label: option } : option
  );

export const SEGMENTED_MAX_OPTIONS = 3;

export const AxisControl = ({ axis, value, onChange, compact }) => {
  const options = normalizeOptions(axis);
  return (
    <div className="flex shrink-0 items-center gap-space-2">
      <Text as="span" variant="xs" color="tertiary">
        {axis.label}
      </Text>
      {!compact && options.length <= SEGMENTED_MAX_OPTIONS ? (
        <GroupedTabs
          size="xs"
          value={value}
          onChange={onChange}
          options={options.map((option) => ({
            value: option.value,
            display: option.label,
          }))}
        />
      ) : (
        <Select
          size="xs"
          value={value}
          options={options}
          onChange={(next) => next && onChange(next)}
          aria-label={axis.label}
          triggerClassName="w-[6.5rem]"
        />
      )}
    </div>
  );
};

export const CommandMenuDemo = ({ showSelectedIndicator }) => {
  const [value, setValue] = useState('new_dataset');
  const wrapperRef = useRef(null);
  // cmdk schedules `scrollIntoView({ block: 'nearest' })` on the selected
  // item after mount, which walks every scroll ancestor and pulls the outer
  // page down to this demo. Patch scrollIntoView to a no-op on every cmdk
  // element inside this wrapper so the scheduled call has no effect.
  useLayoutEffect(() => {
    const wrapper = wrapperRef.current;
    if (!wrapper) return;
    const targets = wrapper.querySelectorAll(
      '[cmdk-item], [cmdk-group-heading], [cmdk-list]'
    );
    const originals = new Map();
    targets.forEach((el) => {
      originals.set(el, el.scrollIntoView);
      el.scrollIntoView = () => {};
    });
    return () => {
      originals.forEach((fn, el) => {
        el.scrollIntoView = fn;
      });
    };
  }, []);
  return (
    <div
      ref={wrapperRef}
      className="max-h-64 w-full max-w-80 overflow-auto rounded-md border border-default bg-elevated"
    >
      <CommandMenu
        value={value}
        onSelect={setValue}
        showSelectedIndicator={showSelectedIndicator === 'shown'}
        placeholder="Search actions…"
        items={[
          { value: 'new_dataset', label: 'New dataset' },
          { value: 'new_prompt', label: 'New prompt' },
          { value: 'new_experiment', label: 'New experiment' },
          { value: 'invite', label: 'Invite a teammate' },
        ]}
      />
    </div>
  );
};

      className="w-full max-w-80"
      items={items}
      layout={layout}
      onItemClick={(item) =>
        setHidden((previous) =>
          previous.includes(item.id)
            ? previous.filter((id) => id !== item.id)

        )
      }
    />
  );
};

export const OverviewPage = () => {
  const [query, setQuery] = useState('');
  const [category, setCategory] = useState('all');
  const normalizedQuery = query.trim().toLowerCase();
  const matches = CATALOG.filter(
    (entry) =>
      (category === 'all' || entry.category === category) &&
      (normalizedQuery === '' ||
        entry.name.toLowerCase().includes(normalizedQuery) ||
        entry.tag.includes(normalizedQuery))
  );
  const visibleCategories = CATEGORIES.filter((group) =>
    matches.some((entry) => entry.category === group.id)
  );
  return (
    <main className="min-h-screen bg-surface-level-1 text-primary">
      <header className="border-b border-subtle bg-brand-subtle-gradient px-space-5 py-space-8 sm:px-space-8">
        <div className="mx-auto flex max-w-[100rem] flex-col items-start gap-space-5">
          <Logo
            brand="langchain"
            variant="full"
            size="lg"
            className="text-primary"
          />
          <div className="flex flex-col gap-space-3">
            <Text as="h1" variant="h1">
              Component overview
            </Text>
            <Text color="secondary" className="max-w-2xl">
              All the components! Each card runs a real instance with its own
              state — click it, type in it, open it, etc.
            </Text>
          </div>
        </div>
      </header>
      <div
        className="sticky top-0 border-b border-subtle bg-surface-level-1/95 px-space-5 py-space-4 backdrop-blur-sm sm:px-space-8"
        style={{ zIndex: zIndices.stickyHeader }}
      >
        <div className="mx-auto flex max-w-[100rem] flex-wrap items-center gap-x-space-4 gap-y-space-3">
          <div className="w-full max-w-64">
            <Input
              size="sm"
              value={query}
              onChange={setQuery}
              debounceMs={0}
              placeholder="Filter components…"
              aria-label="Filter components"
              leftDecorator={<MagnifyingGlassIcon size={16} weight="regular" />}
            />
          </div>
          <div className="flex flex-wrap items-center gap-space-1">
            <Button
              size="xs"
              color={category === 'all' ? 'primary' : 'secondary'}
              variant={category === 'all' ? 'normal' : 'plain'}
              onClick={() => setCategory('all')}
            >
              All
            </Button>
            {CATEGORIES.map((group) => (
              <Button
                key={group.id}
                size="xs"
                color={category === group.id ? 'primary' : 'secondary'}
                variant={category === group.id ? 'normal' : 'plain'}
                onClick={() => setCategory(group.id)}
              >
                {group.label}
              </Button>
            ))}
          </div>
          <Badge color="secondary" size="xs" className="ml-auto">
            {`${matches.length} of ${CATALOG.length}`}
          </Badge>
        </div>
      </div>
      <div className="mx-auto flex max-w-[100rem] flex-col gap-space-8 px-space-5 py-space-8 sm:px-space-8">
        {visibleCategories.map((group) => (
          <section
            key={group.id}
            className="flex flex-col gap-space-4"
            aria-labelledby={`overview-${group.id}`}
          >
            <div className="flex flex-wrap items-baseline gap-x-space-3 gap-y-space-1 border-b border-subtle pb-space-3">

                {group.label}

              <Text variant="sm" color="tertiary">
                {group.blurb}
              </Text>
            </div>

              {matches

                  <OverviewCard key={entry.name} entry={entry} />
                ))}
            </div>
          </section>
        ))}
        {matches.length === 0 && (
          <div className="flex flex-col items-center gap-space-2 py-space-8">
            <Text variant="h3">No components match “{query}”</Text>

          </div>

      </div>
    </main>
  );
